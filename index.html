if (btn.dataset.chartType) chartType = btn.dataset.chartType; if (btn.dataset.chartScope) chartScope = btn.dataset.chartScope; if (btn.dataset.series) { const id = btn.dataset.series; if (hiddenSeries.has(id)) hiddenSeries.delete(id); const modal = modalId ? document.getElementById(modalId) : btn.closest("dialog"); if (modal) closeModal(modal); }); }); elements.toastClose?.addEventListener("click", () => { if (elements.toast) elements.toast.hidden = true;
